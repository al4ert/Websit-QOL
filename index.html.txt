  <label><input type="checkbox" id="h1"> Homework</label>
  <label><input type="checkbox" id="h2"> Move 10 minutes</label>
  <label><input type="checkbox" id="h3"> No phone in bed</label>

  <button type="button" id="reset">Reset today</button>

  <script>
    const key = "today-" + new Date().toDateString();
    document.getElementById("date").textContent = new Date().toDateString();

    ["h1", "h2", "h3"].forEach((id) => {
      const box = document.getElementById(id);
      box.checked = localStorage.getItem(key + id) === "1";
      box.addEventListener("change", () => {
        localStorage.setItem(key + id, box.checked ? "1" : "0");
      });
    });

    document.getElementById("reset").onclick = () => {
      ["h1", "h2", "h3"].forEach((id) => {
        document.getElementById(id).checked = false;
        localStorage.removeItem(key + id);
      });
    };
  </script>
